3. CONSULTA DE DEMOSTRACIÓN (SOLO POSIBLE DESPUÉS DEL CAMBIO):
--------------------------------------------------------------------------------
Comando ejecutado:
docker exec seismic-data-visualization-system_db_1 psql -U postgres -d datawarehouse_copia -c "
SELECT 
    c.id_lote AS lote,
    p.id_prediccion,
    s.id_sismo,
    s.magnitud,
    s.referencia_de_localizacion AS ubicacion,
    s.nombre_estado AS estado,
    p.clase_riesgo_predicha AS riesgo,
    p.score_anomalia,
    p.probabilidad_confianza AS confianza,
    p.version_modelo,
    p.timestamp_prediccion
FROM predicciones_ia_sismos p
JOIN dim_sismos s ON p.id_sismo = s.id_sismo
JOIN control_ingesta_automatica c ON p.id_lote_ingesta = c.id_lote
WHERE p.es_anomalia = TRUE AND p.clase_riesgo_predicha = 'CRITICO'
ORDER BY p.score_anomalia DESC;"
